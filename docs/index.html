Task 1 — Forest share per canton (sf)

Task: Using swissTLM3D (forest) and swissBOUNDARIES3D (cantons), calculate the percentage of area covered by forest per canton and visualize the results (map and plot).
All computations are performed in EPSG:2056 (LV95), so areas are in m².

library(sf)
library(dplyr)
library(ggplot2)
library(scales)
library(tictoc)

sf_use_s2(FALSE)

tlm_path <- "data/SWISSTLM3D_2026_LV95_LN02.gpkg"
bnd_path <- "data/swissBOUNDARIES3D_1_5_LV95_LN02.gpkg"

# Cantons
kantone_raw <- st_read(bnd_path, layer = "tlm_kantonsgebiet", quiet = TRUE)

name_candidates <- c("NAME", "NAME_DE", "KANTONSNAME", "KANTON", "name", "kanton")
name_col <- name_candidates[name_candidates %in% names(kantone_raw)][1]
if (is.na(name_col)) stop("No canton name column detected.")

kantone <- kantone_raw |>
  transmute(name = .data[[name_col]]) |>
  st_zm(drop = TRUE, what = "ZM") |>
  st_make_valid()

if (st_crs(kantone)$epsg != 2056) kantone <- st_transform(kantone, 2056)

kantone$canton_area <- as.numeric(st_area(kantone))

# Forest share per canton
tic("sf forest share")
forest_area_by_canton <- lapply(seq_len(nrow(kantone)), function(i) {
  k <- kantone[i, ]

  idx <- st_intersects(k, wald, sparse = TRUE)[[1]]
  if (length(idx) == 0) {
    return(data.frame(name = k$name, canton_area = k$canton_area, wald_area = 0))
  }

  w_sub <- wald[idx, , drop = FALSE]
  w_sub <- st_make_valid(w_sub)

  inter <- suppressWarnings(st_intersection(w_sub, k))
  inter <- st_collection_extract(inter, "POLYGON")

  w_area <- sum(as.numeric(st_area(inter)), na.rm = TRUE)

  data.frame(name = k$name, canton_area = k$canton_area, wald_area = w_area)
})

Task 2 — DuckDB workflow + timing

Task: Recreate Task 1 using the provided DuckDB database (wald-kantone.duckdb) and measure execution time with tictoc.

library(DBI)
library(duckdb)
library(tictoc)

db_path <- "data/wald-kantone.duckdb"

con <- dbConnect(duckdb(), dbdir = db_path, read_only = TRUE)
try(dbExecute(con, "INSTALL spatial;"), silent = TRUE)
[1] 0
dbExecute(con, "LOAD spatial;")
[1] 0
try(dbExecute(con, "CREATE INDEX kantone_idx ON kantone USING RTREE (geom);"), silent = TRUE)
try(dbExecute(con, "CREATE INDEX wald_idx ON wald USING RTREE (geom);"), silent = TRUE)

duckdb_sql <- "
WITH wald_kantone AS (
  SELECT
    k.name AS name,
    k.area AS canton_area,
    st_area(st_intersection(w.geom, k.geom)) AS wald_area
  FROM wald w, kantone k
  WHERE st_intersects(w.geom, k.geom)
),
wald_kanton_grp AS (
  SELECT
    name,
    canton_area,
    SUM(wald_area) AS wald_area
  FROM wald_kantone
  GROUP BY name, canton_area
)
SELECT
  name,
  canton_area,
  wald_area,
  wald_area / canton_area AS wald_share
FROM wald_kanton_grp
ORDER BY wald_share DESC;
"

tic("DuckDB SQL forest share")
wald_share_duckdb <- dbGetQuery(con, duckdb_sql)
toc()

dbDisconnect(con)

Discussion

The task 1 workflow took ca. 4-5 minutes, while the DuckDB task 2 SQL workflow took ca. 2 and half minutes.

DuckDB was faster because the spatial intersection and aggregation were executed inside the database engine instead of in R memory.

Both methods produce nearly identical results; the main difference is where the computationally expensive spatial overlay is performed (R vs. database engine).

Task 3 — Chessboard moves using DE-9IM

This task identifies possible rook, bishop, and king moves on a chessboard grid using DE-9IM spatial relation patterns.

library(sf)
library(ggplot2)

gpkg <- "data/chessboard.gpkg"

grid_orig <- st_read(gpkg, layer = "grid_orig", quiet = TRUE)
grid_dest <- st_read(gpkg, layer = "grid_dest", quiet = TRUE)

# Assign a valid projected CRS (no transformation, only labeling)
st_crs(grid_orig) <- 3857
st_crs(grid_dest) <- 3857

# DE-9IM patterns
st_rook   <- \(x, y) st_relate(x, y, pattern = "F***1****")  # shared edge
st_bishop <- \(x, y) st_relate(x, y, pattern = "F***0****")  # shared corner
st_king   <- \(x, y) st_relate(x, y, pattern = "F***T****")  # any touch

# Compute moves
rook_moves   <- grid_dest[grid_orig, , op = st_rook]
bishop_moves <- grid_dest[grid_orig, , op = st_bishop]
king_moves   <- grid_dest[grid_orig, , op = st_king]

# Plot function
plot_moves <- function(moves, title_text){
  ggplot() +
    geom_sf(data = grid_dest, fill = "white", color = "grey60") +
    geom_sf(data = moves, fill = "lightgreen", color = "red", linewidth = 1.2) +
    geom_sf(data = grid_orig, fill = "grey80", color = "black", linewidth = 1.2) +
    labs(title = title_text) +
    theme_minimal()
}

plot_moves(rook_moves, "Rook moves")
plot_moves(bishop_moves, "Bishop moves")
plot_moves(king_moves, "King moves")
